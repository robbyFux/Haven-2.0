1. Create core/decorators.py:
   - admin_required: a decorator that wraps @login_required and additionally checks request.user.is_admin. If not admin, return HttpResponseForbidden("Admin access required."). Use functools.wraps.
   ```python
   from functools import wraps
   from django.contrib.auth.decorators import login_required
   from django.http import HttpResponseForbidden

   def admin_required(view_func):
       @wraps(view_func)
       @login_required
       def wrapper(request, *args, **kwargs):
           if not request.user.is_admin:
               return HttpResponseForbidden("Admin access required.")
           return view_func(request, *args, **kwargs)
       return wrapper
   ```

2. Add to admin_panel/views.py:
   - user_detail(request, user_id): @admin_required
     target_user = get_object_or_404(HavenUser, id=user_id)
     events = Event.objects.filter(user_id=target_user.id).count()
     devices = Device.objects.filter(user_id=target_user.id).count()
     form = QuotaEditForm(initial={"storage_quota_mb": target_user.storage_quota_mb, "max_events": target_user.max_events})
     Render user_detail.html with target_user, form, events count, devices count

   - quota_edit(request, user_id): @admin_required, POST only
     target_user = get_object_or_404(HavenUser, id=user_id)
     form = QuotaEditForm(request.POST)
     if form.is_valid():
       target_user.storage_quota_mb = form.cleaned_data["storage_quota_mb"]
       target_user.max_events = form.cleaned_data["max_events"]
       target_user.save(update_fields=["storage_quota_mb", "max_events"])
       messages.success(request, f"Quota updated for {target_user.username}")
       if request.htmx: return render partials/user_row.html with target_user
       return redirect("admin_panel:user_detail", user_id=user_id)
     else:
       if request.htmx: return HttpResponse(form.errors.as_text(), status=422)
       return render user_detail.html with form errors

     target_user = get_object_or_404(HavenUser, id=user_id)
     target_user.is_active = not target_user.is_active
     target_user.save(update_fields=["is_active"])
     messages.success(request, f"{'Activated' if target_user.is_active else 'Deactivated'} {target_user.username}")
     if request.htmx: return render partials/user_row.html with target_user
     return redirect("admin_panel:user_detail", user_id=user_id)
