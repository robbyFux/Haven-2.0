val pushoverChannel = notifChannels.filterIsInstance<PushoverChannel>().firstOrNull()
if (pushoverChannel != null && heartbeatPushoverMin > 0) {
    launch {
        while (_state.value != MonitorState.ACTIVE) delay(500)
        while (true) {
            delay(heartbeatPushoverMin * 60_000L)
            val msg = "Haven alive - v${BuildConfig.VERSION_NAME} - ${_state.value} - ..."

        }
    }
}

<!-- Pushover API: POST https://api.pushover.net/1/messages.json (multipart/form-data) -->
Required fields: token (app token), user (user/group key), message (text ≤1024 chars)
Optional: title, priority (-2..2), attachment (JPEG as ByteArray, field "attachment", content-type "image/jpeg")
Emergency priority (2): requires retry=60, expire=3600
Priority mapping: LOW→0, MEDIUM→0, HIGH→1, CRITICAL→2

Constants:

`send(event, attachment)`:
- Build `MultipartBody` using `MultipartBody.Builder().setType(MultipartBody.FORM)`
- Add parts: `token`, `user`, `title` ("[Haven] Alert"), `message` (buildMessage(event)), `priority` (derived from event.severity)
- Priority: `LOW/MEDIUM → "0"`, `HIGH → "1"`, `CRITICAL → "2"` with additional parts `retry="60"`, `expire="3600"`
- If attachment != null: add `addFormDataPart("attachment", "alert.jpg", attachment.toRequestBody("image/jpeg".toMediaType()))`
- POST to `API_URL`, check `response.isSuccessful`, wrap in `withContext(Dispatchers.IO) { runCatching { ... } }`

`sendHeartbeat(message)`:
- Same POST structure, title = "[Haven] Heartbeat", message = message param, priority = "0", no attachment

`buildMessage(event: TriggerEvent): String`:
- Same format as MattermostChannel: trigger type, severity, timestamp, sensor value (use SimpleDateFormat)

KDoc on class explaining Pushover API, that attachment is a JPEG ByteArray (JPEG only, max 2.5 MB per Pushover limits),
and the emergency priority (CRITICAL) behavior.

Do NOT add jsonStr() — multipart encoding handles escaping automatically.

private fun PushoverConfigDialog(
    initialUserKey: String,
    initialAppToken: String,
    onDismiss: () -> Unit,

                    modifier = Modifier.fillMaxWidth(),
